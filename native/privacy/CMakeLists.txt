# The helper can be built alone without Qt or any third-party dependency.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.24)
  project(LmtiAuditAccess LANGUAGES CXX)
  option(LMTI_BUILD_TESTS "Build audit access tests" ON)
  enable_testing()
endif()
if(WIN32)
  add_library(lmti_audit_access STATIC windows_access.cpp)
  target_compile_features(lmti_audit_access PUBLIC cxx_std_20)
  target_link_libraries(lmti_audit_access PUBLIC advapi32)
  add_executable(lmti-audit-acl audit_acl_main.cpp)
  target_link_libraries(lmti-audit-acl PRIVATE lmti_audit_access)
  # Stable trusted location for the source CLI; install/package uses bin/.
  set_target_properties(lmti-audit-acl PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(lmti-audit-acl PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  if(MSVC)
    foreach(target lmti_audit_access lmti-audit-acl)
      set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    endforeach()
    target_link_options(lmti-audit-acl PRIVATE /DEPENDENTLOADFLAG:0x800)
  endif()
  install(TARGETS lmti-audit-acl RUNTIME DESTINATION bin)
  if(LMTI_BUILD_TESTS)
    add_executable(lmti-audit-acl-tests windows_access_tests.cpp)
    target_link_libraries(lmti-audit-acl-tests PRIVATE lmti_audit_access)
    if(MSVC)
      set_target_properties(lmti-audit-acl-tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      target_compile_options(lmti-audit-acl-tests PRIVATE /W4 /permissive- /utf-8)
    endif()
    add_test(NAME audit_acl_native COMMAND lmti-audit-acl-tests)
    set_tests_properties(audit_acl_native PROPERTIES TIMEOUT 20)
  endif()
endif()
